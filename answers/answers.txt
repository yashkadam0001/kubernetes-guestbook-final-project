    <script>
        const form = document.getElementById('guestbookForm');
        const authorInput = document.getElementById('authorInput');
        const messageInput = document.getElementById('messageInput');
        const entriesList = document.getElementById('entriesList');
        const emptyState = document.getElementById('emptyState');
        const entryCount = document.getElementById('entryCount');

        const initialEntries = [
            {
                author: "Kubernetes Admin",
                message: "Welcome to the Guestbook v1 application running on Kubernetes!",
                timestamp: new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit' })
            }
        ];

        let entries = [];
        try {
            const saved = localStorage.getItem('guestbook_v1_entries');
            entries = saved ? JSON.parse(saved) : initialEntries;
        } catch (e) {
            entries = initialEntries;
        }

        function renderEntries() {
            entryCount.textContent = `${entries.length} ${entries.length === 1 ? 'entry' : 'entries'}`;
            if (entries.length === 0) {
                entriesList.innerHTML = '<li class="empty-state">No entries yet. Be the first to leave a message!</li>';
                return;
            }

            entriesList.innerHTML = entries.map(entry => `
                <li class="entry-item">
                    <div class="entry-header">
                        <span class="entry-author">${escapeHtml(entry.author)}</span>
                        <span class="entry-date">${escapeHtml(entry.timestamp)}</span>
                    </div>
                    <p class="entry-text">${escapeHtml(entry.message)}</p>
                </li>
            `).join('');
        }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        form.addEventListener('submit', function(e) {
            e.preventDefault();
            const author = authorInput.value.trim();
            const message = messageInput.value.trim();

            if (!author || !message) return;

            const newEntry = {
                author: author,
                message: message,
                timestamp: new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric', hour: '2-digit', minute: '2-digit' })
            };

            entries.unshift(newEntry);
            try {
                localStorage.setItem('guestbook_v1_entries', JSON.stringify(entries));
            } catch (err) {}

            authorInput.value = '';
            messageInput.value = '';
            renderEntries();
        });

        renderEntries();
    </script>
